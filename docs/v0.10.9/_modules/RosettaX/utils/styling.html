def merge_style(
    *styles,
) -> dict:
    """
    Merge style dictionaries while ignoring non dictionary values.
    """
    merged_style = {}

    for style in styles:
        if isinstance(style, dict):
            merged_style.update(style)

    return merged_style

def copy_style(
    style,
) -> dict:
    """
    Return a mutable copy of a style dictionary.
    """
    if isinstance(style, dict):
        return dict(style)

    return {}

def build_rgba(
    color_name: str,

) -> str:
