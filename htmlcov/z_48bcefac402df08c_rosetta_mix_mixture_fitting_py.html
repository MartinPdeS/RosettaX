266def centers_in_observed_valleys( 

267 *, 

268 fitted_centers: np.ndarray, 

269 histogram_centers: np.ndarray, 

270 smoothed_counts: np.ndarray, 

271 count_variance: np.ndarray, 

272 observed_modes: np.ndarray, 

273) -> np.ndarray: 

274 """Identify fitted log-intensity centers in statistically resolved valleys. 

275 

276 Parameters 

277 ---------- 

278 fitted_centers : ndarray, shape (k,) 

279 Component centers in log10 intensity. 

280 histogram_centers : ndarray, shape (n,) 

281 Increasing bin centers in log10 intensity. 

282 smoothed_counts : ndarray, shape (n,) 

283 Smoothed event counts per bin. 

284 count_variance : ndarray, shape (n,) 

285 Poisson variance propagated through the smoothing kernel. 

286 observed_modes : ndarray, shape (m,) 

287 Increasing histogram indices of significant observed maxima. 

288 

289 Returns 

290 ------- 

291 ndarray of bool, shape (k,) 

292 True where both bracketing modes exceed the center by 2.5 estimated 

293 standard deviations. This empirical check requires resolved modes. 

294 A single unresolved mode cannot establish that two populations exist. 

295 """ 

296 if observed_modes.size < 2: 

297 return np.zeros(fitted_centers.size, dtype=bool) 

298 mode_centers = histogram_centers[observed_modes] 

299 right_indices = np.searchsorted(mode_centers, fitted_centers) 

300 bracketed = (right_indices > 0) & (right_indices < observed_modes.size) 

301 right = observed_modes[np.clip(right_indices, 1, observed_modes.size - 1)] 

302 left = observed_modes[np.clip(right_indices - 1, 0, observed_modes.size - 2)] 

303 center_counts = np.interp(fitted_centers, histogram_centers, smoothed_counts) 

304 center_variance = np.interp(fitted_centers, histogram_centers, count_variance) 

305 left_noise = np.sqrt(np.maximum(count_variance[left] + center_variance, 1.0)) 

306 right_noise = np.sqrt(np.maximum(count_variance[right] + center_variance, 1.0)) 

307 return bracketed & ( 

308 smoothed_counts[left] - center_counts >= 2.5 * left_noise 

309 ) & (smoothed_counts[right] - center_counts >= 2.5 * right_noise)
